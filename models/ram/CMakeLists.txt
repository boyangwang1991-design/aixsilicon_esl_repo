cmake_minimum_required(VERSION 3.19)
project(AixEslRam VERSION 0.1.0 LANGUAGES CXX)
include("${CMAKE_CURRENT_LIST_DIR}/../../cmake/ESLModel.cmake")
esl_model(ram AixEslRam)
option(ESL_RAM_BUILD_TESTS "Build RAM tests (off for consumers)" OFF)
if(ESL_RAM_BUILD_TESTS)
    enable_testing()
    add_executable(ram_tests tests/systemc/test_ram.cpp)
    target_link_libraries(ram_tests PRIVATE aix::esl::ram)
    foreach(case normal functional counters_off reset drain capacity invalid unbound)
        add_test(NAME ram_${case} COMMAND ram_tests ${case})
        set_tests_properties(ram_${case} PROPERTIES TIMEOUT 15)
    endforeach()
endif()
