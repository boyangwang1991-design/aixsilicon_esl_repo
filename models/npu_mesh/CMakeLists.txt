cmake_minimum_required(VERSION 3.19)
project(AixEslNpuMesh VERSION 0.1.0 LANGUAGES CXX)
include("${CMAKE_CURRENT_LIST_DIR}/../../cmake/ESLModel.cmake")
if(NOT TARGET aix::esl::npu_sram_controller)
  add_subdirectory(../npu_sram_controller ${CMAKE_CURRENT_BINARY_DIR}/sram-library)
endif()
esl_model(npu_mesh AixEslNpuMesh LIBRARIES aix::esl::npu_sram_controller PACKAGES AixEslNpuSramController)
target_sources(aix_esl_npu_mesh PRIVATE systemc/src/network.cpp systemc/src/dma.cpp systemc/src/axi.cpp systemc/src/target.cpp)
option(ESL_MESH_BUILD_TESTS "Build mesh verification" OFF)
option(ESL_MESH_BUILD_RUNNER "Build mesh trace runner" OFF)
if(ESL_MESH_BUILD_TESTS)
  enable_testing()
  add_executable(mesh_qos_tests tests/systemc/test_qos.cpp)
  target_link_libraries(mesh_qos_tests PRIVATE aix::esl::npu_mesh)
  target_include_directories(mesh_qos_tests PRIVATE ../../services/systemc/include)
  foreach(case shape shape_reset fifo priority aged)
    add_test(NAME mesh_qos_${case} COMMAND mesh_qos_tests ${case})
    set_tests_properties(mesh_qos_${case} PROPERTIES TIMEOUT 30)
  endforeach()
  add_executable(mesh_tests tests/systemc/test_model.cpp)
  target_link_libraries(mesh_tests PRIVATE aix::esl::npu_mesh)
  add_executable(mesh_network_tests tests/systemc/test_network.cpp)
  target_link_libraries(mesh_network_tests PRIVATE aix::esl::npu_mesh)
  add_test(NAME mesh_network_bounds COMMAND mesh_network_tests)
  add_executable(mesh_memory_tests tests/systemc/test_memory.cpp)
  target_link_libraries(mesh_memory_tests PRIVATE aix::esl::npu_mesh)
  foreach(case sram_data sram_time window bank ddr multicast_large reset_native partial cancel dma_reset slow_fanout reservation timeout_native)
    add_test(NAME mesh_memory_${case} COMMAND mesh_memory_tests ${case})
    set_tests_properties(mesh_memory_${case} PROPERTIES TIMEOUT 30)
  endforeach()
  foreach(case data ordering backpressure dma multicast reset timeout timeout_tail dma_cancel token command config split central axi management)
    add_test(NAME mesh_${case} COMMAND mesh_tests ${case})
    set_tests_properties(mesh_${case} PROPERTIES TIMEOUT 30)
  endforeach()
endif()
if(ESL_MESH_BUILD_RUNNER)
  add_executable(mesh_run systemc/src/runner.cpp)
  target_link_libraries(mesh_run PRIVATE aix::esl::npu_mesh)
endif()
