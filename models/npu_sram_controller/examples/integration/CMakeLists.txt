cmake_minimum_required(VERSION 3.19)
project(NpuSramExternalConsumer LANGUAGES CXX)
if(ESL_SOURCE)
  set(ESL_NPU_BUILD_RUNNER OFF CACHE BOOL "" FORCE)
  set(ESL_NPU_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  add_subdirectory(${ESL_SOURCE}/models/npu_sram_controller npu-library)
else()
  find_package(AixEslNpuSramController 0.1 CONFIG REQUIRED)
endif()
add_executable(npu_consumer main.cpp)
target_link_libraries(npu_consumer PRIVATE aix::esl::npu_sram_controller)
enable_testing()
add_test(NAME independent_sram_consumer COMMAND npu_consumer)
set_tests_properties(independent_sram_consumer PROPERTIES TIMEOUT 30)
