cmake_minimum_required(VERSION 3.19)
project(AixEslNpuSramController VERSION 0.1.0 LANGUAGES CXX)
include("${CMAKE_CURRENT_LIST_DIR}/../../cmake/ESLModel.cmake")
esl_model(npu_sram_controller AixEslNpuSramController)
target_sources(aix_esl_npu_sram_controller PRIVATE systemc/src/mapping.cpp)
option(ESL_NPU_BUILD_TESTS "Build SRAM unit tests" OFF)
option(ESL_NPU_BUILD_RUNNER "Build DAG benchmark runner" ON)
if(ESL_NPU_BUILD_TESTS)
  enable_testing()
  add_executable(npu_sram_tests tests/systemc/test_model.cpp)
  target_link_libraries(npu_sram_tests PRIVATE aix::esl::npu_sram_controller)
  foreach(case mapping data timing backpressure ecc hierarchy ordering negative reset pipeline partial_race dual hol aw_w)
    add_test(NAME npu_${case} COMMAND npu_sram_tests ${case})
    set_tests_properties(npu_${case} PROPERTIES TIMEOUT 30)
  endforeach()
endif()
if(ESL_NPU_BUILD_RUNNER)
  add_executable(npu_sram_run systemc/src/runner.cpp)
  target_link_libraries(npu_sram_run PRIVATE aix::esl::npu_sram_controller)
endif()
