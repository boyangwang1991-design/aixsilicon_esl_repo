cmake_minimum_required(VERSION 3.19)
project(min_systemc LANGUAGES CXX)
include("${CMAKE_CURRENT_LIST_DIR}/../../../cmake/ESLSystemC.cmake")
add_executable(min_sc main.cpp)
target_compile_features(min_sc PRIVATE cxx_std_${ESL_CXX_STANDARD})
target_link_libraries(min_sc PRIVATE SystemC::systemc)
add_executable(min_tlm tlm_smoke.cpp)
target_compile_features(min_tlm PRIVATE cxx_std_${ESL_CXX_STANDARD})
target_link_libraries(min_tlm PRIVATE SystemC::systemc)
enable_testing()
add_test(NAME systemc_counter COMMAND min_sc)
add_test(NAME tlm_transaction COMMAND min_tlm)
set_tests_properties(systemc_counter tlm_transaction PROPERTIES TIMEOUT 15)
