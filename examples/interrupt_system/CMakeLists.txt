cmake_minimum_required(VERSION 3.19)
project(AixEslInterruptSystem LANGUAGES CXX)
if(ESL_MODELS_SOURCE_DIR)
    foreach(model host_master tlm_bus timer irq_controller)
        if(NOT TARGET aix::esl::${model})
            add_subdirectory("${ESL_MODELS_SOURCE_DIR}/models/${model}" "models/${model}")
        endif()
    endforeach()
else()
    foreach(package AixEslHostMaster AixEslTlmBus AixEslTimer AixEslIrqController)
        find_package(${package} 0.1 CONFIG REQUIRED)
    endforeach()
endif()
add_executable(interrupt_system systemc/main.cpp)
target_link_libraries(interrupt_system PRIVATE aix::esl::host_master aix::esl::tlm_bus aix::esl::timer aix::esl::irq_controller)
enable_testing()
foreach(case oneshot periodic reprogram irq reset capacity errors transport unbound unbound_irq invalid_config invalid_timer)
    add_test(NAME interrupt_${case} COMMAND interrupt_system ${case})
    set_tests_properties(interrupt_${case} PROPERTIES TIMEOUT 15)
endforeach()
