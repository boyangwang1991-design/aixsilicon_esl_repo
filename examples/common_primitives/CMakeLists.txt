cmake_minimum_required(VERSION 3.19)
project(AixEslCommonConsumer LANGUAGES CXX)
if(ESL_MODELS_SOURCE_DIR)
    if(NOT TARGET aix::esl::common)
        add_subdirectory("${ESL_MODELS_SOURCE_DIR}/common" common)
    endif()
else()
    find_package(AixEslCommon 0.1 CONFIG REQUIRED)
endif()
add_executable(common_primitives systemc/main.cpp)
target_link_libraries(common_primitives PRIVATE aix::esl::common)
enable_testing()
foreach(case queue queue_clear queue_events queue_copy_error gate gate_exception gate_events store store_errors metrics metrics_off overflow invalid)
    add_test(NAME common_${case} COMMAND common_primitives ${case})
    set_tests_properties(common_${case} PROPERTIES TIMEOUT 15)
endforeach()
add_executable(common_architecture systemc/architecture.cpp)
target_link_libraries(common_architecture PRIVATE aix::esl::common)
foreach(case pipeline resource_instances mapping arbitration transactions ordering random)
    add_test(NAME common_${case} COMMAND common_architecture ${case})
    set_tests_properties(common_${case} PROPERTIES TIMEOUT 15)
endforeach()
add_executable(common_services systemc/services.cpp)
target_link_libraries(common_services PRIVATE aix::esl::common)
foreach(case registers ecc dag)
    add_test(NAME common_${case} COMMAND common_services ${case})
    set_tests_properties(common_${case} PROPERTIES TIMEOUT 15)
endforeach()
add_executable(common_composition systemc/composition.cpp)
target_link_libraries(common_composition PRIVATE aix::esl::common)
foreach(mapping stripe xor)
    foreach(observation off trace)
        add_test(NAME common_composition_${mapping}_${observation}
            COMMAND common_composition ${mapping} ${observation} 8 1)
        set_tests_properties(common_composition_${mapping}_${observation} PROPERTIES TIMEOUT 15)
    endforeach()
endforeach()
add_test(NAME common_composition_serial COMMAND common_composition stripe off 8 8)
add_test(NAME common_composition_fault COMMAND common_composition stripe off 8 1 fault)
set_tests_properties(common_composition_serial common_composition_fault PROPERTIES TIMEOUT 15)
add_executable(common_infrastructure systemc/infrastructure.cpp)
target_link_libraries(common_infrastructure PRIVATE aix::esl::common)
foreach(case channel cdc lifecycle sparse events traffic checkers faults)
    add_test(NAME common_${case} COMMAND common_infrastructure ${case})
    set_tests_properties(common_${case} PROPERTIES TIMEOUT 15)
endforeach()
add_executable(common_task_pipeline systemc/task_pipeline.cpp)
target_link_libraries(common_task_pipeline PRIVATE aix::esl::common)
add_test(NAME common_task_pipeline COMMAND common_task_pipeline 64 pass)
add_test(NAME common_task_failure COMMAND common_task_pipeline 64 fail)
set_tests_properties(common_task_pipeline common_task_failure PROPERTIES TIMEOUT 15)
add_executable(common_storage_reliability systemc/storage_reliability.cpp)
target_link_libraries(common_storage_reliability PRIVATE aix::esl::common)
foreach(case dense sparse timed_dense timed_sparse timed_baseline)
    add_test(NAME common_storage_${case} COMMAND common_storage_reliability ${case})
    set_tests_properties(common_storage_${case} PROPERTIES TIMEOUT 15)
endforeach()
add_executable(common_policies systemc/policies.cpp)
target_link_libraries(common_policies PRIVATE aix::esl::common)
foreach(case compute interrupts)
    add_test(NAME common_policy_${case} COMMAND common_policies ${case})
    set_tests_properties(common_policy_${case} PROPERTIES TIMEOUT 15)
endforeach()
