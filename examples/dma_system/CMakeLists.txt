cmake_minimum_required(VERSION 3.19)
project(AixEslDmaSystem LANGUAGES CXX)
if(ESL_MODELS_SOURCE_DIR)
    foreach(model host_master tlm_bus ram rom dma)
        if(NOT TARGET aix::esl::${model})
            add_subdirectory("${ESL_MODELS_SOURCE_DIR}/models/${model}" "models/${model}")
        endif()
    endforeach()
else()
    foreach(package AixEslHostMaster AixEslTlmBus AixEslRam AixEslRom AixEslDma)
        find_package(${package} 0.1 CONFIG REQUIRED)
    endforeach()
endif()
add_executable(dma_system systemc/main.cpp)
target_link_libraries(dma_system PRIVATE aix::esl::host_master aix::esl::tlm_bus aix::esl::ram aix::esl::rom aix::esl::dma)
enable_testing()
foreach(case copy tail capacity drain reset_read reset_write reset_queued read_error write_error partial_error invalid dual contention annotation downstream_full invalid_config unbound)
    add_test(NAME dma_${case} COMMAND dma_system ${case})
    set_tests_properties(dma_${case} PROPERTIES TIMEOUT 15)
endforeach()
