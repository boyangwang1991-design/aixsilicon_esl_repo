cmake_minimum_required(VERSION 3.19)
project(AixEslPeripheralSystem LANGUAGES CXX)
if(ESL_MODELS_SOURCE_DIR)
    foreach(model host_master tlm_bus irq_controller uart gpio)
        if(NOT TARGET aix::esl::${model})
            add_subdirectory("${ESL_MODELS_SOURCE_DIR}/models/${model}" "models/${model}")
        endif()
    endforeach()
else()
    foreach(package AixEslHostMaster AixEslTlmBus AixEslIrqController AixEslUart AixEslGpio)
        find_package(${package} 0.1 CONFIG REQUIRED)
    endforeach()
endif()
add_executable(peripheral_system systemc/main.cpp)
target_link_libraries(peripheral_system PRIVATE aix::esl::host_master aix::esl::tlm_bus aix::esl::irq_controller aix::esl::uart aix::esl::gpio)
enable_testing()
foreach(case uart_tx loopback rx_overrun backpressure reset_tx reset_mmio drain_mmio capacity gpio_edges gpio_collision gpio_reset transport errors dual uart_collision gpio_mmio_reset gpio_mmio_drain gpio_capacity unbound_uart unbound_gpio invalid_uart invalid_gpio)
    add_test(NAME peripheral_${case} COMMAND peripheral_system ${case})
    set_tests_properties(peripheral_${case} PROPERTIES TIMEOUT 15)
endforeach()
