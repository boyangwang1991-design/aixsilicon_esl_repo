cmake_minimum_required(VERSION 3.19)
project(RegisterConsumer LANGUAGES CXX)
if(ESL_MODEL_SOURCE_DIR)
    add_subdirectory("${ESL_MODEL_SOURCE_DIR}" model)
else()
    find_package({{package}} 0.1 CONFIG REQUIRED)
endif()
add_executable(consumer main.cpp)
target_link_libraries(consumer PRIVATE aix::esl::{{name}})
enable_testing()
add_test(NAME register_behavior COMMAND consumer)
add_test(NAME register_unbound COMMAND consumer unbound)
set_tests_properties(register_behavior register_unbound PROPERTIES TIMEOUT 15)
